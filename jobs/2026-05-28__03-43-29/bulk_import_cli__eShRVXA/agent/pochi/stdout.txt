add_and_log() {
  local text="$1"
  local event_json
  event_json=$(mem0 add "$text" --user-id "$USER_ID" --output json)
  local event_id
  event_id=$(printf '%s' "$event_json" | jq -r '.event_id')
  while true; do
    local status_json
    status_json=$(mem0 event status "$event_id" --output json)
    local status
    status=$(printf '%s' "$status_json" | jq -r '.data.status')
    if [ "$status" = "SUCCEEDED" ]; then
      local memory_id
      memory_id=$(printf '%s' "$status_json" | jq -r '.data.results[0].id')
      echo "Added memory: $memory_id" >> "$LOG"
      break
    fi
    sleep 1
  done
}
